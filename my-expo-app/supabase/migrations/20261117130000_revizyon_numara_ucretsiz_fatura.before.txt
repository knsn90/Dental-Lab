                   FROM order_items oi
                  WHERE (oi.work_order_id = wo.id)
                  GROUP BY COALESCE(oi.currency, ls.default_currency, 'TRY'::text)) t), '{}'::jsonb) AS totals_by_currency

     LEFT JOIN clinics c ON ((c.id = d.clinic_id)))
