-- ==== FUNC _field_staff_sees(uuid,uuid) ====
CREATE OR REPLACE FUNCTION public._field_staff_sees(p_wo uuid, p_assigned uuid)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select case public.my_role_key()
    when 'technician' then
      p_assigned = auth.uid()
      or exists (select 1 from public.order_stages os
                  where os.work_order_id = p_wo
                    and (os.technician_id = auth.uid()
                         or (os.technician_id is null and os.status = 'aktif')))
    when 'courier' then
      exists (select 1 from public.deliveries d
               where d.work_order_id = p_wo and d.courier_id = auth.uid())
    else true
  end
$function$


-- ==== FUNC can_see_work_order_id(uuid) ====
CREATE OR REPLACE FUNCTION public.can_see_work_order_id(p_wo uuid)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select exists (select 1 from public.work_orders wo
                  where wo.id = p_wo and wo.lab_id = public.get_my_lab_id()
                    and public._field_staff_sees(wo.id, wo.assigned_to))
$function$

-- ==== FUNC doctor_owns_order_doctor(uuid) ====

 RETURNS boolean

      or public._doctor_profile_owns_doctor(auth.uid(), p_doctor_id);
$function$


-- ==== FUNC get_display_tech_names(uuid[]) ====
CREATE OR REPLACE FUNCTION public.get_display_tech_names(p_ids uuid[])
 RETURNS TABLE(id uuid, name text)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  SELECT p.id,
    CASE
      WHEN (SELECT user_type FROM profiles WHERE id = auth.uid())
           IN ('doctor','clinic_admin','clinic_secretary')
      THEN CASE COALESCE(ls.technician_name_visibility, 'full')
             WHEN 'hidden' THEN ''
             WHEN 'first'  THEN split_part(COALESCE(p.full_name, ''), ' ', 1)
             ELSE COALESCE(p.full_name, '')
           END
      ELSE COALESCE(p.full_name, '')
    END AS name
  FROM profiles p
  LEFT JOIN lab_settings ls ON ls.lab_id = p.lab_id
  WHERE p.id = ANY(p_ids);
$function$


-- ==== FUNC is_admin_user() ====
CREATE OR REPLACE FUNCTION public.is_admin_user()
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  SELECT EXISTS (
    SELECT 1 FROM profiles
    WHERE id = auth.uid()
      AND user_type = 'admin'
  );
$function$

-- ==== FUNC lab_can_view_finance() ====
CREATE OR REPLACE FUNCTION public.lab_can_view_finance()
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  select public.is_platform_admin() or (
    exists (select 1 from public.profiles where id = auth.uid() and user_type in ('admin', 'lab'))
    and (public.has_permission('view_financials') or public.has_permission('view_finance')
         or public.has_permission('manage_finance')))
$function$


-- ==== FUNC log_activity(text,text,uuid,text,jsonb,uuid,uuid) ====
CREATE OR REPLACE FUNCTION public.log_activity(p_action text, p_entity_type text DEFAULT NULL::text, p_entity_id uuid DEFAULT NULL::uuid, p_entity_label text DEFAULT NULL::text, p_metadata jsonb DEFAULT NULL::jsonb, p_lab_id uuid DEFAULT NULL::uuid, p_actor_id uuid DEFAULT NULL::uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_authed uuid := auth.uid();
  v_actor  uuid;
  v_name text; v_ut text; v_role text; v_plab uuid; v_id uuid;
BEGIN
  IF p_action IS NULL OR btrim(p_action) = '' THEN RETURN NULL; END IF;

  -- Authenticated caller: actor is ALWAYS the JWT subject (cannot be forged).
  -- Trusted server context (auth.uid() IS NULL: triggers / service_role / cron): may name the actor.
  IF v_authed IS NOT NULL THEN
    v_actor := v_authed;
  ELSE
    v_actor := p_actor_id;
  END IF;

  SELECT full_name, user_type, role, lab_id INTO v_name, v_ut, v_role, v_plab
    FROM public.profiles WHERE id = v_actor;

  INSERT INTO public.activity_logs
    (actor_id, actor_name, actor_type, action, entity_type, entity_id, entity_label, metadata, lab_id)
  VALUES (v_actor, COALESCE(NULLIF(v_name,''),'Sistem'),
     public._log_actor_type(v_ut, v_role),
     p_action, p_entity_type, p_entity_id, p_entity_label, p_metadata,
     -- authenticated: lab is derived from the actor's own profile (p_lab_id ignored);
     -- trusted context: honor p_lab_id, else the actor's profile lab.
     CASE WHEN v_authed IS NOT NULL THEN v_plab ELSE COALESCE(p_lab_id, v_plab) END)
  RETURNING id INTO v_id;
  RETURN v_id;
EXCEPTION WHEN OTHERS THEN RETURN NULL;
END; $function$


-- ==== FUNC my_clinic_id() ====
CREATE OR REPLACE FUNCTION public.my_clinic_id()
 RETURNS uuid
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
  SELECT clinic_id FROM profiles WHERE id = auth.uid() LIMIT 1;
$function$
